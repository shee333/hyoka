class Program

    {
        try
        {
            Console.WriteLine("Azure Resource Group Management Demo");
            Console.WriteLine("=====================================\n");

            // 1. Authenticate using DefaultAzureCredential
            Console.WriteLine("1. Authenticating with Azure...");
            var credential = new DefaultAzureCredential();
            var armClient = new ArmClient(credential);
            
            // Get the default subscription
            var subscription = await armClient.GetDefaultSubscriptionAsync();
            Console.WriteLine($"   ✓ Connected to subscription: {subscription.Data.DisplayName} ({subscription.Data.SubscriptionId})");
            Console.WriteLine();

            // 2. Create a new resource group
            string resourceGroupName = $"rg-demo-{Guid.NewGuid().ToString()[..8]}";
            string location = "eastus";
            
            Console.WriteLine($"2. Creating resource group '{resourceGroupName}' in {location}...");
            var resourceGroupData = new ResourceGroupData(location)
            {
                Tags =
                {
                    { "Environment", "Demo" },
                    { "CreatedBy", "AzureSDK" }
                }
            };

            var resourceGroupCollection = subscription.GetResourceGroups();
            var resourceGroupResponse = await resourceGroupCollection.CreateOrUpdateAsync(
                WaitUntil.Completed, 
                resourceGroupName, 
                resourceGroupData
            );
            var resourceGroup = resourceGroupResponse.Value;
            
            Console.WriteLine($"   ✓ Resource group created successfully");
            Console.WriteLine($"   - Name: {resourceGroup.Data.Name}");
            Console.WriteLine($"   - Location: {resourceGroup.Data.Location}");
            Console.WriteLine($"   - Provisioning State: {resourceGroup.Data.ProvisioningState}");
            Console.WriteLine();

            // 3. List all resource groups in the subscription
            Console.WriteLine("3. Listing all resource groups in subscription...");
            var resourceGroups = resourceGroupCollection.GetAllAsync();
            int count = 0;
            
            await foreach (var rg in resourceGroups)
            {
                count++;
                Console.WriteLine($"   - {rg.Data.Name} ({rg.Data.Location})");
            }
            Console.WriteLine($"   ✓ Total resource groups: {count}");
            Console.WriteLine();

            // 4. Get details of the created resource group
            Console.WriteLine($"4. Getting details of resource group '{resourceGroupName}'...");
            var retrievedResourceGroup = await resourceGroupCollection.GetAsync(resourceGroupName);
            var rgData = retrievedResourceGroup.Value.Data;
            
            Console.WriteLine($"   ✓ Resource Group Details:");
            Console.WriteLine($"   - ID: {rgData.Id}");
            Console.WriteLine($"   - Name: {rgData.Name}");

            var updatedResourceGroup = updateResponse.Value;

# Azure Resource Group Manager

A C# console application demonstrating Azure Resource Group management using the Azure.ResourceManager SDK (Management Plane).

## Features

This application demonstrates:
1. ✅ Authentication using `DefaultAzureCredential`
2. ✅ Creating a new resource group in the "eastus" region
3. ✅ Listing all resource groups in the subscription
4. ✅ Getting details of a specific resource group
5. ✅ Adding tags to a resource group
6. ✅ Deleting a resource group

## Required NuGet Packages

```xml
<PackageReference Include="Azure.Identity" Version="1.11.0" />
<PackageReference Include="Azure.ResourceManager" Version="1.11.1" />
```

- **Azure.Identity**: Provides Azure Active Directory token authentication support
- **Azure.ResourceManager**: The new unified Azure management SDK

## Prerequisites

### Authentication Options

The application uses `DefaultAzureCredential`, which tries these authentication methods in order:

1. **Environment variables** (Service Principal):
   ```bash
   set AZURE_CLIENT_ID=<your-client-id>
   set AZURE_CLIENT_SECRET=<your-client-secret>
   set AZURE_TENANT_ID=<your-tenant-id>
   ```

2. **Azure CLI**: Log in using:
   ```bash
   az login
   ```

3. **Azure PowerShell**: Log in using:
   ```powershell
   Connect-AzAccount
   ```

4. **Managed Identity** (when running on Azure resources)

5. **Visual Studio** or **VS Code** credentials

### Azure Permissions

Your account/service principal needs:
- **Contributor** or **Owner** role on the subscription
- Or custom role with these permissions:
  - `Microsoft.Resources/subscriptions/resourceGroups/read`
  - `Microsoft.Resources/subscriptions/resourceGroups/write`
  - `Microsoft.Resources/subscriptions/resourceGroups/delete`

## How to Build and Run

### 1. Restore dependencies:

dotnet restore

### 2. Build the project:

dotnet build
